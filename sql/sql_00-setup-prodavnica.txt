-- Redosled brisanja je obrnut od redosleda kreiranja, zbog stranih kljuceva.
-- Ne mozes obrisati kupci dok porudzbine pokazuju na njih.
DROP TABLE IF EXISTS stavke_porudzbine;
DROP TABLE IF EXISTS porudzbine;
DROP TABLE IF EXISTS proizvodi;
DROP TABLE IF EXISTS kupci;


CREATE TABLE kupci (
    id SERIAL PRIMARY KEY,
    ime TEXT NOT NULL,
    grad TEXT,
    datum_registracije DATE
);

CREATE TABLE proizvodi (
    id SERIAL PRIMARY KEY,
    naziv TEXT NOT NULL,
    kategorija TEXT,
    cena NUMERIC(10,2)
);

CREATE TABLE porudzbine (
    id SERIAL PRIMARY KEY,
    kupac_id INTEGER REFERENCES kupci(id),
    datum DATE,
    status TEXT
);

CREATE TABLE stavke_porudzbine (
    id SERIAL PRIMARY KEY,
    porudzbina_id INTEGER REFERENCES porudzbine(id),
    proizvod_id INTEGER REFERENCES proizvodi(id),
    kolicina INTEGER
);
